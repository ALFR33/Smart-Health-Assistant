<!DOCTYPE html>
<html lang="ar" dir="rtl">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Sahih Assistant - مساعد صحيح الذكي</title>
  <!-- Tailwind CSS CDN -->
  <script src="https://cdn.tailwindcss.com"></script>
  <!-- React & ReactDOM CDN -->
  <script src="https://unpkg.com/react@18/umd/react.production.min.js" crossorigin></script>
  <script src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js" crossorigin></script>
  <!-- Babel Standalone for JSX -->
  <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>
  <!-- FontAwesome Icons -->
  <link rel="stylesheet" href="https://cdnjs.cloudflare.com/ajax/libs/font-awesome/6.4.0/css/all.min.css">
  <style>
    body { font-family: system-ui, -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif; }
  </style>
</head>
<body class="bg-slate-950 text-slate-100 min-h-screen flex items-center justify-center p-2<div id="root" class="w-full max-w-4xl"></div>

  <script type="text/babel">
    const { useState, useRef, useEffect } = React;

    function SahihAssistant() {
      const [activeTab, setActiveTab] = useState("chat");
      const [messages, setMessages] = useState([
        {
          id: "1",
          sender: "sahih",
          text: "أهلاً بك! أنا **صحيح (Sahih)**، مساعدك الذكي الشامل. كيف يمكنني مساعدتك اليوم؟ يمكنك كتابة مقالات عن الفوائد، مناقشة الأفكار، أو إنتاج موسيقى حتى 3 دقائق!",
          time: new Date().toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" }),
        }
      ]);
      const [inputText, setInputText] = useState("");
      const [isTyping, setIsTyping] = useState(false);
      const [musicPrompt, setMusicPrompt] = useState("");
      const [songDuration, setSongDuration] = useState("3:00");
      const [isGenerating, setIsGenerating] = useState(false);

      const messagesEndRef = useRef(null);

      useEffect(() => {
        messagesEndRef.current?.scrollIntoView({ behavior: "smooth" });
      }, [messages, isTyping]);

      const handleSend = (textToSend) => {
        const query = textToSend || inputText;
        if (!query.trim()) return;

        const userMsg = {
          id: Date.now().toString(),
          text: query,
          time: new Date().toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" }),
        };

        setMessages((prev) => [...prev, userMsg]);
        if (!textToSend) setInputText("");
        setIsTyping(true);

        setTimeout(() => {
          let responseText = "";
          const lower = query.toLowerCase();

          if (lower.includes("موسيقى") || lower.includes("أغنية")) {
            responseText = "صحيح في خدمتك! يمكنك التوجه لتبويب 'صانع الموسيقى' لإنتاج هذا اللحن فوراً حتى 3 دقائق كاملة.";
          } else {
            const topic = query.replace(/(فوائد|فائدة|ماهي|ما هي|عن|مقال|أكتب|اكتب)/g, "").trim() || query;
            responseText = `📝 **مقال شامل أعده مساعدك (صحيح) عن: ${topic}**\n\n` +
              `**+ من الخيارات القيمة التي تمنح الإنسان مزايا صحية ويومية عديدة.\n\n` +
              `**أبرز الفوائد الرئيسية:**\n` +
              `1. **تعزيز الصحة والمناعة:** غنية بالعناصر المغذية ومضادات الأكسدة.\n` +
              `2. **دعم وظائف الجسم:** تمنح الجسم توازناً ونشاطاً ممتازاً.\n` +
              `3. **الوقاية اليومية:** تسهم في نمط حياة صحي ومستدام.\n\n` +
              `💡 *يمكن لـ صحيح تحويل هذا الموضوع إلى مقطع موسيقي يصل حتى 3 دقائق!*`;
          }

          setMessages((prev) => [
            ...prev,
            {
              id: (Date.now() + 1).toString(),
              sender: "sahih",
              text: responseText,
              time: new Date().toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" }),
            }
          ]);
          setIsTyping(false);
        }, 1000);
      };

      const handleGenerateMusic = () => {
        if (!musicPrompt.trim()) return;
        setIsGenerating(true);
        setTimeout(() => {
          setMessages((prev) => [
            ...prev,
            {
              id: Date.now().toString(),
              sender: "sahih",
              text: `تم إنتاج الموسيقى بنجاح بطول يصل إلى (${songDuration}) بناءً على طلبك: "${musicPrompt}"!`,
              time: new Date().toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" }),
            }
          ]);
          setIsGenerating(false);
          setActiveTab("chat");
        }, 2000);
      };

      return (
        <div className="h-[85vh] bg-slate-900 border border-slate-800 rounded-2xl shadow-2xl flex flex-col overflow-hidden">
          {/* Header */}
          <header className="px-6 py-4 bg-slate-900/90 border-b border-slate-800 flex items-center justify-between">
            <div className="flex items-center gap-3">
              <div className="w-10 h-10 rounded-xl bg-gradient-to-tr from-emerald-500 to-teal-400 flex items-center justify-center text-slate-950 font-bold text-lg">
                <i className="fa-solid fa-shield-halved"></i>
              </div>
              <div>
                <h1 className="font-bold text-lg text-emerald-400">صحيح | Sahih Assistant</h1>
                <p className="text-xs text-slate-400">مساعد ذكي للإجابات، المقالات، والإنتاج الموسيقي</p>
              </div>
            </div>
            <span className="px-3 py-1 text-xs bg-emerald-500/10 text-emerald-400 border border-emerald-500/20 rounded-full">
              نشط بلا حدود
            </span>
          </header>

          {/* Navigation */}
          <nav className="flex items-center justify-around bg-slate-900/50 border-b border-slate-800 p-2">
            <button
              onClick={() => setActiveTab("chat")}
              className={`px-4 py-2 text-sm font-medium rounded-xl transition ${
                activeTab === "chat" ? "bg-emerald-500/10 text-emerald-400 border border-emerald-500/30" : "text-slate-400"
              }`}
            >
              <i className="fa-regular fa-comments ml-2"></i> المحادثة
            </button>
            <button
              onClick={() => setActiveTab("music")}
              className={`px-4 py-2 text-sm font-medium rounded-xl transition ${
                activeTab === "music" ? "bg-emerald-500/10 text-emerald-400 border border-emerald-500/30" : "text-slate-400"
              }`}
            >
              <i className="fa-solid fa-music ml-2"></i> صانع الموسيقى
            </button>
            <button
              onClick={() => setActiveTab("ideas")}
              className={`px-4 py-2 text-sm font-medium rounded-xl transition ${
                activeTab === "ideas" ? "bg-emerald-500/10 text-emerald-400 border border-emerald-500/30" : "text-slate-400"
              }`}
            >
              <i className="fa-regular fa-lightbulb ml-2"></i> الأفكار
            </button>
          </nav>

          {/* Body */}
          <div className="flex-1 overflow-y-auto p-6 space-y-4">
            {activeTab === "chat" && (
              <div className="flex flex-col h-full justify-between">
                <div className="flex-1 overflow-y-auto space-y-4">
                  {messages.map((msg) => (
                    <div key={msg.id} className={`flex items-start gap-3 ${msg.sender === "user" ? "flex-row-reverse" : ""}`}>
                      <div className={`w-8 h-8 rounded-lg flex items-center justify-center shrink-0 ${msg.sender === "user" ? "bg-indigo-600" : "bg-emerald-500 text-slate-950 font-bold"}`}>
                        {msg.sender === "user" ? <i className="fa-solid fa-user"></i> : <i className="fa-solid fa-robot"></i>}
                      </div>
                      <div className={`max-w-[80%] rounded-2xl p-4 text-sm ${msg.sender === "user" ? "bg-indigo-600/20 text-indigo-100 border border-indigo-500/30" : "bg-slate-800 text-slate-200 border border-slate-700"}`}>
                        <div className="whitespace-pre-line">{msg.text}</div>
                        <span className="text-[10px] text-slate-500 mt-1 block">{msg.time}</span>
                      </div>
                    </div>
                  ))}
                  {isTyping && <div className="text-slate-400 text-xs py-2">صحيح يكتب الإجابة...</div>}
                  <div ref={messagesEndRef} />
                </div>

                <div className="pt-4">
                  <div className="flex bg-slate-800 border border-slate-700 rounded-xl overflow-hidden">
                    <input
                      type="text"
                      value={inputText}
                      onChange={(e) => setInputText(e.target.value)}
                      onKeyDown={(e) => e.key === "Enter" && handleSend()}
                      placeholder="اسأل صحيح أي شيء، أو اطلب مقالاً عن الفوائد..."
                      className="w-full px-4 py-3 bg-transparent text-sm text-slate-100 focus:outline-none"
                    />
                    <button onClick={() => handleSend()} className="px-5 bg-emerald-500 hover:bg-emerald-400 text-slate-950 font-bold">
                      <i className="fa-solid fa-paper-plane"></i>
                    </button>
                  </div>
                </div>
              </div>
            )}

            {activeTab === "music" && (
              <div className="space-y-4 max-w-lg mx-auto py-4">
                <h2 className="text-lg font-bold text-emerald-400 text-center">إنتاج الموسيقى بحجم حتى 3 دقائق</h2>
                <div className="bg-slate-800 p-5 rounded-2xl border border-slate-700 space-y-4">
                  <div>
                    <label className="block text-xs font-semibold mb-1">وصف الصوت والأسلوب</label>
                    <textarea
                      rows="3"
                      value={musicPrompt}
                      onChange={(e) => setMusicPrompt(e.target.value)}
                      placeholder="مثال: لحن بيانو سينمائي هادئ مع إيقاع لوفي"
                      className="w-full p-3 bg-slate-900 border border-slate-700 rounded-xl text-sm focus:outline-none"
                    />
                  </div>
                  <div>
                    <label className="block text-xs font-semibold mb-1">مدة الموسيقى (حتى 3 دقائق)</label>
                    <select
                      value={songDuration}
                      onChange={(e) => setSongDuration(e.target.value)}
                      className="w-full p-3 bg-slate-900 border border-slate-700 rounded-xl text-sm focus:outline-none"
                    >
                      <option value="3:00">3 دقائق كاملة (3:00)</option>
                      <option value="2:00">دقيقتان (2:00)</option>
                      <option value="1:00">دقيقة واحدة (1:00)</option>
                    </select>
                  </div>
                  <button
                    onClick={handleGenerateMusic}
                    disabled={isGenerating || !musicPrompt.trim()}
                    className="w-full py-3 bg-emerald-500 text-slate-950 font-bold rounded-xl hover:bg-emerald-400 transition"
                  >
                    {isGenerating ? "جاري الإنتاج..." : "إنتاج اللحن الآن"}
                  </button>
                </div>
              </div>
            )}

            {activeTab === "ideas" && (
              <div className="space-y-4">
                <h2 className="text-base font-bold text-emerald-400">أفكار مقترحة</h2>
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                  <div className="p-4 bg-slate-800 border border-slate-700 rounded-xl">
                    <h3 className="font-bold text-sm">مقال عن فوائد العسل</h3>
                    <p className="text-xs text-slate-400 mt-1">توليد مقال شامل يحتوي على أهم الفوائد الصحية.</p>
                    <button onClick={() => { handleSend("فوائد العسل"); setActiveTab("chat"); }} className="mt-3 text-xs bg-emerald-500/20 text-emerald-400 px-3 py-1 rounded-lg">طلب المقال</button>
                  </div>
                  <div className="p-4 bg-slate-800 border border-slate-700 rounded-xl">
                    <h3 className="font-bold text-sm">لحن استرخاء بطول 3 دقائق</h3>
                    <p className="text-xs text-slate-400 mt-1">إنتاج مقطع بيانو لوفي بطول 3 دقائق.</p>
                    <button onClick={() => { setMusicPrompt("relaxing lo-fi piano"); setActiveTab("music"); }} className="mt-3 text-xs bg-emerald-500/20 text-emerald-400 px-3 py-1 rounded-lg">توليد الموسيقى</button>
                  </div>
                </div>
              </div>
            )}
          </div>
        </div>
      );
    }

    ReactDOM.createRoot(document.getElementById("root")).render(<SahihAssistant />);
  </script>
</body>
</html>
